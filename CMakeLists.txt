cmake_minimum_required(VERSION 3.20)
project(klc_cpp LANGUAGES CXX)

option(KLC_BIG_N "Use 128-bit packing (maximum n=24, GCC/Clang)" OFF)
option(KLC_NATIVE "Optimize for the build machine, enabling its SIMD instructions" ON)
option(KLC_SCALAR "Disable explicit SIMD, including on native builds" OFF)
option(KLC_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)

add_library(klc_core src/Polynomial.cpp src/Permutation.cpp src/Bruhat.cpp
                     src/KLCalculator.cpp src/Lascoux.cpp)
target_compile_features(klc_core PUBLIC cxx_std_20)
target_include_directories(klc_core PUBLIC include)
if(KLC_BIG_N)
    target_compile_definitions(klc_core PUBLIC USE_128BIT_PACKING)
endif()
if(KLC_SCALAR)
    target_compile_definitions(klc_core PUBLIC KLC_DISABLE_SIMD)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(klc_core PUBLIC -Wall -Wextra)
    if(KLC_NATIVE)
        target_compile_options(klc_core PUBLIC -march=native -mtune=native
                               $<$<CONFIG:Release>:-funroll-loops>)
    endif()
    if(KLC_SANITIZERS)
        target_compile_options(klc_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(klc_core PUBLIC -fsanitize=address,undefined)
    endif()
elseif(MSVC)
    target_compile_options(klc_core PUBLIC /W4)
    if(KLC_BIG_N)
        message(FATAL_ERROR "KLC_BIG_N requires a compiler with __uint128_t")
    endif()
endif()

add_executable(klc_cpp main.cpp src/Cli.cpp src/Commands.cpp)
target_link_libraries(klc_cpp PRIVATE klc_core)

include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE klc_core)
    add_test(NAME core_tests COMMAND core_tests)
    add_test(NAME verify COMMAND klc_cpp --verify "${CMAKE_CURRENT_SOURCE_DIR}/verify.txt")
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME cli_tests COMMAND "${Python3_EXECUTABLE}"
                 "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py" "$<TARGET_FILE:klc_cpp>")
        add_test(NAME tools_tests COMMAND "${Python3_EXECUTABLE}"
                 "${CMAKE_CURRENT_SOURCE_DIR}/tests/tools_tests.py")
    endif()
endif()
